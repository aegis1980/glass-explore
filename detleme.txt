clientside_callback(
    f"""
    function(clickData, figure, activeTab) {{
        if (!figure) {{
            return window.dash_clientside.no_update;
        }}

        // No click yet: leave as-is
        if (!clickData || !clickData.points || clickData.points.length === 0) {{
            return window.dash_clientside.no_update;
        }}

        const pt = clickData.points[0];

        const x = pt.x;
        const y = pt.y;
        if pt.z !== undefined {{
            //const z = pt.z;
        }}
        const color = pt['marker.color'];

        // Make a shallow copy so Dash notices the update
        const newFig = {{
            ...figure,
            data: figure.data.map(trace => ({{...trace}}))
        }};

        // Find the dedicated 'selected_id' trace by name
        const highlightIndex = newFig.data.findIndex(
            trace => trace.name === "selected_id"
        );

        if (highlightIndex === -1) {{
            return window.dash_clientside.no_update;
        }}

        console.log(color);
        newFig.data[highlightIndex].x = [x];
        newFig.data[highlightIndex].y = [y];
        if pt.z !== undefined {{
           // newFig.data[highlightIndex].z = [z];
        }}
        newFig.data[highlightIndex].marker.color = [color];

        return newFig;
    }}
    """,
    Output(EnergyLayoutID.GRAPH_IGDB, "figure", allow_duplicate=True),
    Input(EnergyLayoutID.GRAPH_IGDB, "clickData"),
    State(EnergyLayoutID.GRAPH_IGDB, "figure"),
    State(EnergyLayoutID.TABS,"active_tab" ),
    prevent_initial_call=True
)
